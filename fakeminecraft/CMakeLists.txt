cmake_minimum_required(VERSION 3.25)
project(Stonefield LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_GLSLANG_BINARIES ON CACHE BOOL "" FORCE)
set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(vulkan_headers URL https://github.com/KhronosGroup/Vulkan-Headers/archive/refs/tags/v1.4.304.tar.gz)
FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz)
FetchContent_Declare(glm URL https://github.com/g-truc/glm/archive/refs/tags/1.0.1.tar.gz)
FetchContent_Declare(vma URL https://github.com/GPUOpen-LibrariesAndSDKs/VulkanMemoryAllocator/archive/refs/tags/v3.3.0.tar.gz)
FetchContent_Declare(volk URL https://github.com/zeux/volk/archive/refs/tags/1.4.304.tar.gz)
FetchContent_Declare(glslang URL https://github.com/KhronosGroup/glslang/archive/refs/tags/15.4.0.tar.gz)
FetchContent_MakeAvailable(vulkan_headers glfw glm vma volk glslang)
target_include_directories(volk PUBLIC ${vulkan_headers_SOURCE_DIR}/include)
add_executable(stonefield src/main.cpp src/world.cpp src/world.hpp)
target_include_directories(stonefield PRIVATE ${vulkan_headers_SOURCE_DIR}/include ${vma_SOURCE_DIR}/include ${volk_SOURCE_DIR} ${glm_SOURCE_DIR})
target_compile_definitions(stonefield PRIVATE GLFW_INCLUDE_NONE VK_NO_PROTOTYPES GLM_FORCE_DEPTH_ZERO_TO_ONE)
target_link_libraries(stonefield PRIVATE glfw volk)
set_target_properties(stonefield PROPERTIES WIN32_EXECUTABLE FALSE)
foreach(shader terrain.vert terrain.frag ui.vert ui.frag)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${shader}.spv
    COMMAND $<TARGET_FILE:glslang-standalone> -V ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader} -o ${CMAKE_CURRENT_BINARY_DIR}/${shader}.spv
    DEPENDS glslang-standalone ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader})
  list(APPEND SHADER_OUTPUTS ${CMAKE_CURRENT_BINARY_DIR}/${shader}.spv)
endforeach()
add_custom_target(shaders ALL DEPENDS ${SHADER_OUTPUTS})
add_dependencies(stonefield shaders)
add_custom_command(TARGET stonefield POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${SHADER_OUTPUTS} $<TARGET_FILE_DIR:stonefield>)
add_custom_command(TARGET stonefield POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:stonefield>/assets)
